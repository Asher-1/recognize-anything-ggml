cmake_minimum_required(VERSION 3.21)
project(recognize_anything_ggml LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

option(RAM_GGML_CUDA "Build ggml CUDA backend" OFF)
option(RAM_GGML_VULKAN "Build ggml Vulkan backend" OFF)
if(RAM_GGML_CUDA AND RAM_GGML_VULKAN)
    message(FATAL_ERROR "Use separate build directories for CUDA and Vulkan")
endif()

set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_CUDA ${RAM_GGML_CUDA} CACHE BOOL "" FORCE)
set(GGML_VULKAN ${RAM_GGML_VULKAN} CACHE BOOL "" FORCE)
if(RAM_GGML_CUDA AND NOT DEFINED GGML_CUDA_GRAPHS)
    set(GGML_CUDA_GRAPHS ON CACHE BOOL "Enable CUDA Graphs for repeated inference")
endif()
set(GGML_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/ggml")
if(NOT EXISTS "${GGML_SOURCE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "Initialize ggml with: git submodule update --init cpp_ggml/third_party/ggml")
endif()
execute_process(COMMAND git -C "${GGML_SOURCE_DIR}" rev-parse HEAD
    OUTPUT_VARIABLE GGML_HEAD OUTPUT_STRIP_TRAILING_WHITESPACE
    RESULT_VARIABLE GGML_GIT_RESULT)
if(NOT GGML_GIT_RESULT EQUAL 0 OR NOT GGML_HEAD STREQUAL "8599e0ea3756c4bac4ef813af2241cb1a8bbfb0b")
    message(FATAL_ERROR "ggml must be pinned to v0.21.0 (8599e0ea): ${GGML_HEAD}")
endif()

file(GLOB GGML_PATCHES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/patches/*.patch")
list(SORT GGML_PATCHES)
foreach(GGML_PATCH IN LISTS GGML_PATCHES)
    execute_process(COMMAND git -C "${GGML_SOURCE_DIR}" apply --reverse --check "${GGML_PATCH}"
        RESULT_VARIABLE PATCH_ALREADY_APPLIED OUTPUT_QUIET ERROR_QUIET)
    if(NOT PATCH_ALREADY_APPLIED EQUAL 0)
        execute_process(COMMAND git -C "${GGML_SOURCE_DIR}" apply --check "${GGML_PATCH}"
            RESULT_VARIABLE PATCH_CHECK OUTPUT_VARIABLE PATCH_OUTPUT ERROR_VARIABLE PATCH_ERROR)
        if(NOT PATCH_CHECK EQUAL 0)
            message(FATAL_ERROR "Cannot apply ${GGML_PATCH}: ${PATCH_OUTPUT} ${PATCH_ERROR}")
        endif()
        execute_process(COMMAND git -C "${GGML_SOURCE_DIR}" apply "${GGML_PATCH}"
            RESULT_VARIABLE PATCH_RESULT)
        if(NOT PATCH_RESULT EQUAL 0)
            message(FATAL_ERROR "Failed to apply ${GGML_PATCH}")
        endif()
    endif()
endforeach()

add_subdirectory(third_party/ggml EXCLUDE_FROM_ALL)
find_package(OpenCV REQUIRED COMPONENTS core imgcodecs imgproc)
find_package(OpenMP QUIET COMPONENTS CXX)
set(RAM_GGML_COMMON_LIBS ggml ${OpenCV_LIBS})
add_library(ram-ggml-api STATIC src/main.cpp)
target_compile_definitions(ram-ggml-api PRIVATE RAM_GGML_NO_MAIN)
target_link_libraries(ram-ggml-api PUBLIC ${RAM_GGML_COMMON_LIBS})
target_include_directories(ram-ggml-api PUBLIC ${OpenCV_INCLUDE_DIRS}
    ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_executable(ram-ggml src/main.cpp)
target_link_libraries(ram-ggml PRIVATE ${RAM_GGML_COMMON_LIBS})
target_include_directories(ram-ggml PRIVATE ${OpenCV_INCLUDE_DIRS}
    ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(OpenMP_CXX_FOUND)
    target_link_libraries(ram-ggml-api PUBLIC OpenMP::OpenMP_CXX)
    target_link_libraries(ram-ggml PRIVATE OpenMP::OpenMP_CXX)
endif()
